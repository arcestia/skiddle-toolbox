/* ═══════════════════════════════════════════════════════════════
   Skiddle Toolbox — Shared Theme, Settings & UI Sync
   ═══════════════════════════════════════════════════════════════
   Include this file on every toolbox page to keep the Catppuccin
   theme, settings, and common UI components synchronised.

   - Reads runtime configuration from #tb-config (injected by the
     server) so themes and layouts can be added without editing JS.
   - Reads/writes the chosen theme to localStorage under `toolbox-theme`.
   - Auto-injects a theme switcher into any element with
     `data-tb-theme-bar`.
   - Keeps theme in sync across open tabs via the `storage` event.
   - Tracks recently used tools in localStorage under `toolbox-recent`.
   - Provides a global settings modal.
   - Exposes `window.toolbox` for programmatic use.
   ═══════════════════════════════════════════════════════════════ */

(function () {
  'use strict';

  const THEME_KEY = 'toolbox-theme';
  const SETTINGS_KEY = 'toolbox-settings';
  const LAYOUT_KEY = 'toolbox-layout';
  const ACCENT_KEY = 'toolbox-accent';
  const RADIUS_KEY = 'toolbox-radius';
  const DENSITY_KEY = 'toolbox-density';
  const RECENT_KEY = 'toolbox-recent';

  function loadConfig() {
    try {
      const el = document.getElementById('tb-config');
      if (el) return JSON.parse(el.textContent || '{}');
    } catch (e) {}
    return {
      themes: [{ id: 'mocha', name: 'Mocha', swatch: '#1e1e2e' }],
      layouts: [{ id: 'default', name: 'Default', description: 'Default layout' }],
      accents: [{ id: 'mauve', name: 'Mauve', hue: 267 }],
      radii: [{ id: 'rounded', name: 'Rounded' }],
      densities: [
        { id: 'compact', name: 'Compact', description: 'Tighter spacing' },
        { id: 'comfortable', name: 'Comfortable', description: 'Balanced spacing' },
        { id: 'spacious', name: 'Spacious', description: 'More breathing room' },
      ],
      defaultTheme: 'mocha',
      defaultLayout: 'default',
      defaultAccent: 'mauve',
      defaultRadius: 'rounded',
      defaultDensity: 'comfortable',
    };
  }

  const config = loadConfig();
  const THEMES = config.themes || [];
  const LAYOUTS = config.layouts || [];
  const ACCENTS = config.accents || [];
  const RADII = config.radii || [];
  const DENSITIES = config.densities || [];
  const THEME_IDS = THEMES.map(t => t.id);
  const LAYOUT_IDS = LAYOUTS.map(l => l.id);
  const ACCENT_IDS = ACCENTS.map(a => a.id);
  const RADIUS_IDS = RADII.map(r => r.id);
  const DENSITY_IDS = DENSITIES.map(d => d.id);
  const DEFAULT_THEME = config.defaultTheme || THEME_IDS[0] || 'mocha';
  const DEFAULT_LAYOUT = config.defaultLayout || LAYOUT_IDS[0] || 'default';
  const DEFAULT_ACCENT = config.defaultAccent || ACCENT_IDS[0] || 'mauve';
  const DEFAULT_RADIUS = config.defaultRadius || RADIUS_IDS[0] || 'rounded';
  const DEFAULT_DENSITY = config.defaultDensity || DENSITY_IDS[0] || 'comfortable';

  function getStoredTheme() {
    const raw = localStorage.getItem(THEME_KEY);
    return THEME_IDS.includes(raw) ? raw : DEFAULT_THEME;
  }

  function applyTheme(themeId) {
    if (!THEME_IDS.includes(themeId)) return;
    document.documentElement.setAttribute('data-theme', themeId);
    localStorage.setItem(THEME_KEY, themeId);
    updateActiveButtons(themeId);
    updateThemeSelect(themeId);
  }

  function updateActiveButtons(themeId) {
    document.querySelectorAll('.tb-theme-btn').forEach(btn => {
      const active = btn.dataset.theme === themeId;
      btn.classList.toggle('active', active);
      btn.setAttribute('aria-pressed', active ? 'true' : 'false');
    });
  }

  function themeButton(theme, variant) {
    const btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'tb-theme-btn';
    btn.dataset.theme = theme.id;
    btn.setAttribute('aria-label', theme.name + ' theme');
    btn.setAttribute('aria-pressed', 'false');

    const dot = document.createElement('span');
    dot.className = 'tb-theme-dot';
    dot.style.backgroundColor = theme.swatch;

    if (variant === 'pills') {
      btn.appendChild(dot);
      btn.appendChild(document.createTextNode(theme.name));
    } else {
      btn.appendChild(dot);
    }

    btn.addEventListener('click', () => applyTheme(theme.id));
    return btn;
  }

  function injectThemeBars() {
    document.querySelectorAll('[data-tb-theme-bar]').forEach(container => {
      const variant = container.dataset.tbThemeBar === 'pills' ? 'pills' : 'dots';
      container.classList.add('tb-theme-bar', variant);
      container.innerHTML = '';
      container.setAttribute('role', 'group');
      container.setAttribute('aria-label', 'Theme switcher');
      THEMES.forEach(theme => container.appendChild(themeButton(theme, variant)));
    });
  }

  function initThemeSelect() {
    const select = document.getElementById('tb-theme-select');
    if (!select) return;
    select.innerHTML = '';

    const groups = new Map();
    THEMES.forEach(theme => {
      const pack = theme.pack || 'Other';
      if (!groups.has(pack)) groups.set(pack, []);
      groups.get(pack).push(theme);
    });

    groups.forEach((packThemes, pack) => {
      const optgroup = document.createElement('optgroup');
      optgroup.label = pack;
      packThemes.forEach(theme => {
        const option = document.createElement('option');
        option.value = theme.id;
        option.textContent = theme.name;
        optgroup.appendChild(option);
      });
      select.appendChild(optgroup);
    });
  }

  function updateThemeSelect(themeId) {
    const select = document.getElementById('tb-theme-select');
    const swatch = document.getElementById('tb-theme-swatch');
    if (select) select.value = themeId;
    const theme = THEMES.find(t => t.id === themeId);
    if (swatch && theme) swatch.style.backgroundColor = theme.swatch;
  }

  function getStoredLayout() {
    const raw = localStorage.getItem(LAYOUT_KEY);
    return LAYOUT_IDS.includes(raw) ? raw : DEFAULT_LAYOUT;
  }

  function applyLayout(layoutId) {
    if (!LAYOUT_IDS.includes(layoutId)) return;
    const html = document.documentElement;
    LAYOUT_IDS.forEach(id => html.classList.remove('tb-layout-' + id));
    html.classList.add('tb-layout-' + layoutId);
    localStorage.setItem(LAYOUT_KEY, layoutId);
    updateLayoutButtons(layoutId);
  }

  function updateLayoutButtons(layoutId) {
    document.querySelectorAll('.tb-layout-option').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.layout === layoutId);
    });
  }

  function setLayout(layoutId) {
    applyLayout(layoutId);
  }

  function getStoredAccent() {
    const raw = localStorage.getItem(ACCENT_KEY);
    return ACCENT_IDS.includes(raw) ? raw : DEFAULT_ACCENT;
  }

  function applyAccent(accentId) {
    if (!ACCENT_IDS.includes(accentId)) return;
    document.documentElement.setAttribute('data-accent', accentId);
    localStorage.setItem(ACCENT_KEY, accentId);
    updateAccentButtons(accentId);
  }

  function updateAccentButtons(accentId) {
    document.querySelectorAll('.tb-accent-option').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.accent === accentId);
    });
  }

  function setAccent(accentId) {
    applyAccent(accentId);
  }

  function getStoredRadius() {
    const raw = localStorage.getItem(RADIUS_KEY);
    return RADIUS_IDS.includes(raw) ? raw : DEFAULT_RADIUS;
  }

  function applyRadius(radiusId) {
    if (!RADIUS_IDS.includes(radiusId)) return;
    document.documentElement.setAttribute('data-radius', radiusId);
    localStorage.setItem(RADIUS_KEY, radiusId);
    updateRadiusButtons(radiusId);
  }

  function updateRadiusButtons(radiusId) {
    document.querySelectorAll('.tb-radius-option').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.radius === radiusId);
    });
  }

  function setRadius(radiusId) {
    applyRadius(radiusId);
  }

  function getStoredDensity() {
    let raw = localStorage.getItem(DENSITY_KEY);
    if (!raw) {
      const settings = getSettings();
      if (settings.compact) raw = 'compact';
    }
    return DENSITY_IDS.includes(raw) ? raw : DEFAULT_DENSITY;
  }

  function applyDensity(densityId) {
    if (!DENSITY_IDS.includes(densityId)) return;
    document.body.setAttribute('data-density', densityId);
    localStorage.setItem(DENSITY_KEY, densityId);
    updateDensityButtons(densityId);
  }

  function updateDensityButtons(densityId) {
    document.querySelectorAll('.tb-density-option').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.density === densityId);
    });
  }

  function setDensity(densityId) {
    applyDensity(densityId);
  }

  function getSettings() {
    try {
      const raw = localStorage.getItem(SETTINGS_KEY);
      return raw ? JSON.parse(raw) : {};
    } catch (e) {
      return {};
    }
  }

  function setSettings(settings) {
    localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
  }

  function applyCompactMode(enabled) {
    applyDensity(enabled ? 'compact' : 'comfortable');
  }

  function loadSettings() {
    applyDensity(getStoredDensity());
    applyLayout(getStoredLayout());
  }

  function openSettings() {
    loadSettings();
    document.getElementById('tb-settings-overlay').classList.remove('tb-hidden');
    document.getElementById('tb-settings-modal').classList.remove('tb-hidden');
  }

  function closeSettings() {
    document.getElementById('tb-settings-overlay').classList.add('tb-hidden');
    document.getElementById('tb-settings-modal').classList.add('tb-hidden');
  }

  function toggleCompact(enabled) {
    setDensity(enabled ? 'compact' : 'comfortable');
  }

  function getRecentTools() {
    try {
      const raw = localStorage.getItem(RECENT_KEY);
      const arr = raw ? JSON.parse(raw) : [];
      return Array.isArray(arr) ? arr : [];
    } catch (e) {
      return [];
    }
  }

  function setRecentTools(tools) {
    localStorage.setItem(RECENT_KEY, JSON.stringify(tools.slice(0, 6)));
  }

  function recordToolUse(tool) {
    if (!tool || !tool.href) return;
    const recent = getRecentTools().filter(t => t.href !== tool.href);
    recent.unshift({ href: tool.href, title: tool.title, icon: tool.icon || '🧰' });
    setRecentTools(recent);
  }

  function clearRecentTools() {
    localStorage.removeItem(RECENT_KEY);
    document.querySelectorAll('[data-recent-tools]').forEach(el => {
      el.innerHTML = '<p class="tb-muted" style="font-size:0.85rem;">No recent tools yet.</p>';
    });
  }

  function renderRecentTools(container) {
    const recent = getRecentTools();
    if (recent.length === 0) {
      container.innerHTML = '<p class="tb-muted" style="font-size:0.85rem;">No recent tools yet.</p>';
      return;
    }
    let html = '<div class="recent-tools-grid">';
    recent.forEach(tool => {
      html += '<a href="' + tool.href + '" class="tb-back-link recent-tool-link">' +
        '<span class="recent-tool-icon">' + (tool.icon || '🧰') + '</span>' +
        '<span>' + tool.title + '</span>' +
        '</a>';
    });
    html += '</div>';
    container.innerHTML = html;
  }

  function initRecentTools() {
    document.querySelectorAll('[data-recent-tools]').forEach(container => {
      renderRecentTools(container);
    });

    document.querySelectorAll('[data-tool-card] a[href^="/"]').forEach(link => {
      link.addEventListener('click', () => {
        const card = link.closest('[data-tool-card]');
        if (!card) return;
        recordToolUse({
          href: link.getAttribute('href'),
          title: card.dataset.toolTitle || link.innerText,
          icon: card.dataset.toolIcon || '🧰'
        });
      });
    });
  }

  function init() {
    const theme = getStoredTheme();
    document.documentElement.setAttribute('data-theme', theme);
    injectThemeBars();
    updateActiveButtons(theme);
    initThemeSelect();
    updateThemeSelect(theme);
    applyAccent(getStoredAccent());
    applyRadius(getStoredRadius());
    applyDensity(getStoredDensity());
    loadSettings();
    initRecentTools();

    // Auto-update footer year.
    document.querySelectorAll('#footer-year').forEach(el => {
      el.textContent = String(new Date().getFullYear());
    });

    // Keep tabs in sync when localStorage changes elsewhere.
    window.addEventListener('storage', event => {
      if (event.key === THEME_KEY && THEME_IDS.includes(event.newValue)) {
        applyTheme(event.newValue);
      }
      if (event.key === LAYOUT_KEY && LAYOUT_IDS.includes(event.newValue)) {
        applyLayout(event.newValue);
      }
      if (event.key === ACCENT_KEY && ACCENT_IDS.includes(event.newValue)) {
        applyAccent(event.newValue);
      }
      if (event.key === RADIUS_KEY && RADIUS_IDS.includes(event.newValue)) {
        applyRadius(event.newValue);
      }
      if (event.key === DENSITY_KEY && DENSITY_IDS.includes(event.newValue)) {
        applyDensity(event.newValue);
      }
      if (event.key === SETTINGS_KEY) {
        loadSettings();
      }
    });

    // Close settings on Escape.
    document.addEventListener('keydown', event => {
      if (event.key === 'Escape') closeSettings();
    });

    // Expose a tiny public API.
    window.toolbox = {
      setTheme: applyTheme,
      getTheme: getStoredTheme,
      themes: THEMES,
      setLayout: setLayout,
      getLayout: getStoredLayout,
      layouts: LAYOUTS,
      setAccent: setAccent,
      getAccent: getStoredAccent,
      accents: ACCENTS,
      setRadius: setRadius,
      getRadius: getStoredRadius,
      radii: RADII,
      setDensity: setDensity,
      getDensity: getStoredDensity,
      densities: DENSITIES,
      openSettings: openSettings,
      closeSettings: closeSettings,
      toggleCompact: toggleCompact,
      clearRecentTools: clearRecentTools,
      recordToolUse: recordToolUse
    };
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
